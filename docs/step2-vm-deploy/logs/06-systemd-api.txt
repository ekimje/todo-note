# 할 일 노트 API (애플리케이션 티어) — VM에 직접 설치할 때 쓰는 systemd 서비스
#   설치: sudo cp deploy/todo-api.service /etc/systemd/system/ && sudo systemctl enable --now todo-api
#   DB 접속 정보는 /etc/todo-note/api.env (root만 읽기, 저장소에 넣지 않음)
[Unit]
Description=Todo Note API (FastAPI)
After=network-online.target postgresql.service
Wants=network-online.target
Requires=postgresql.service

[Service]
User=todo-api
Group=todo-api
WorkingDirectory=/opt/todo-note/api
EnvironmentFile=/etc/todo-note/api.env
# nginx만 접근하므로 루프백에만 연다
ExecStart=/opt/todo-note/venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000 --proxy-headers
Restart=on-failure
RestartSec=3
NoNewPrivileges=true
ProtectSystem=strict
ProtectHome=true
PrivateTmp=true

[Install]
WantedBy=multi-user.target
